\[
  \mathrm{
    \begin{aligned}
      E_K &= \frac{m_b}{2} \, \Bigl( \frac{l_w^2}{4} \, ( \dot{\theta}_{w_{left}} + \dot{\theta}_{w_{right}} )^2 + l_b^2 \, \bigl( \dot{\theta}_b^2 + \\
          & \quad + \frac{l_w^2}{L^2} \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} )^2 \, sin^2(\theta_b) \bigr) \Bigr) + \\
          & \quad + \frac{m_b \, l_b \, l_w}{2} \, ( \dot{\theta}_{w_{left}}  + \dot{\theta}_{w_{right}} ) \, \dot{\theta}_b \, cos(\theta_b)  + \frac{I_{b_{yy}}}{2} \, \dot{\theta}_b^2 + \\
          & \quad + \frac{l_w^2}{2 \, L^2} \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} )^2 \, \bigl( I_{b_{xx}} \, sin^2(\theta_b) + I_{b_{zz}} \, cos^2(\theta_b) \bigr) + \frac{m_w \, l_w^2}{2} \, ( \dot{\theta}_{w_{left}}^2 + \dot{\theta}_{w_{right}}^2 ) + \\
          & \quad + \frac{I_{w_{xx}}}{2} \, ( \dot{\theta}_{w_{left}}^2 + \dot{\theta}_{w_{right}}^2 ) + \frac{I_{w_{zz}} \, l_w^2}{L^2} \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} )^2
    \end{aligned}
  }

    E_P = m_b \, g \, ( l_b \, cos(\theta_b) + l_w ) + 2 \, m_w \, g \, l_w

\[
  \mathrm{
    \begin{aligned}
      & \partial_{t \, , \, \dot{\theta}_b} \mathcal{L} - \partial_{\theta_b} \mathcal{L} = Q_{\theta_b} \Rightarrow \\
      & \quad m_b \, l_b^2 \, \ddot{\theta}_b + \frac{m_b \, l_b \, l_w}{2} \, ( \ddot{\theta}_{w_{left}} + \ddot{\theta}_{w_{right}} ) \, cos(\theta_b) +  I_{b_{yy}} \, \ddot{\theta}_b + \\
      & \quad - \frac{m_b \, l_b^2 \, l_w^2}{L^2} \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} )^2 \, sin(\theta_b) \, cos(\theta_b) + \\
      & \quad + \frac{l_w^2}{L^2} \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} )^2 \, ( I_{b_{zz}} - I_{b_{xx}} ) \, sin(\theta_b) \, cos(\theta_b) - m_b \, g \, l_b \, sin(\theta_b) = Q_{\theta_b}
    \end{aligned}
  }
\]

\[
  \mathrm{
    \begin{aligned}
      & \partial_{t \, , \, \dot{\theta}_{w_{left}}} \mathcal{L} - \partial_{\theta_{w_{left}}} \mathcal{L} = Q_{\theta_{w_{left}}} \Rightarrow \\
      & \quad m_b \, l_w^2 \, \Bigl( \frac{1}{4} \, ( \ddot{\theta}_{w_{left}} + \ddot{\theta}_{w_{right}} ) - \frac{l_b^2}{L^2} \, \bigl( ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) \, sin^2(\theta_b) + \\
      & \quad + 2 \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} ) \, \dot{\theta}_b \, sin(\theta_b) \, cos(\theta_b) \bigl) \Bigr) + \frac{m_b \, l_b \, l_w}{2} \, \bigl( \ddot{\theta}_b \, cos(\theta_b) - \dot{\theta_b}^2 \, sin(\theta_b) \bigr) + \\
      & \quad - \frac{l_w^2}{L^2} \, \Bigl( ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) \, \bigl( I_{b_{xx}} \, sin^2(\theta_b) + I_{b_{zz}} \, cos^2(\theta_b) \bigr) + 2 \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} ) \, ( I_{b_{xx}} - I_{b_{zz}} ) \\
      & \quad \dot{\theta}_b \, sin(\theta_b) \, cos(\theta_b) \Bigr) + m_w \, l_w^2 \, \ddot{\theta}_{w_{left}} + I_{w_xx} \, \ddot{\theta}_{w_{left}} - \frac{2 \, I_{w_{zz}} \, l_w^2}{L^2} \, ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) = Q_{\theta_{w_{left}}}
    \end{aligned}
  }
\]

\[
  \mathrm{
    \begin{aligned}
      & \partial_{t \, , \, \dot{\theta}_{w_{right}}} \mathcal{L} - \partial_{\theta_{w_{right}}} \mathcal{L} = Q_{\theta_{w_{right}}} \Rightarrow \\
      & \quad m_b \, l_w^2 \, \Bigl( \frac{1}{4} \, ( \ddot{\theta}_{w_{left}} + \ddot{\theta}_{w_{right}} ) + \frac{l_b^2}{L^2} \, \bigl( ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) \, sin^2(\theta_b) + \\
      & \quad + 2 \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} ) \, \dot{\theta}_b \, sin(\theta_b) \, cos(\theta_b) \bigl) \Bigr) + \frac{m_b \, l_b \, l_w}{2} \, \bigl( \ddot{\theta}_b \, cos(\theta_b) - \dot{\theta_b}^2 \, sin(\theta_b) \bigr) + \\
      & \quad + \frac{l_w^2}{L^2} \, \Bigl( ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) \, \bigl( I_{b_{xx}} \, sin^2(\theta_b) + I_{b_{zz}} \, cos^2(\theta_b) \bigr) + 2 \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_{w_{left}} ) \, ( I_{b_{xx}} - I_{b_{zz}} ) \\
      & \quad \dot{\theta}_b \, sin(\theta_b) \, cos(\theta_b) \Bigr) + m_w \, l_w^2 \, \ddot{\theta}_{w_{right}} + I_{w_xx} \, \ddot{\theta}_{w_{right}} + \frac{2 \, I_{w_{zz}} \, l_w^2}{L^2} \, ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) = Q_{\theta_{w_{right}}}
    \end{aligned}
  }

\[
  \mathrm{
    \begin{aligned}
      Q_{\theta_{w_{left}}} &= T_{m_{left}} - T_{f_{left}} + \frac{l_w}{2} \, F \\
      &= \frac{K_t}{R_m} \, v_{left} - \frac{K_e \, K_t}{R_m} \, ( \dot{\theta}_{w_{left}} - \dot{\theta}_b ) - b_f \, ( \dot{\theta}_{w_{left}} - \dot{\theta}_b ) + \frac{l_w}{2} \, F \\
      Q_{\theta_{w_{right}}} &= T_{m_{right}}-T_{f_{right}} + \frac{l_w}{2} \, F \\
      &= \frac{K_t}{R_m} \, v_{right} - \frac{K_e \, K_t}{R_m} \,( \dot{\theta}_{w_{right}} - \dot{\theta}_b ) - b_f \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_b ) + \frac{l_w}{2} \, F \\
      Q_{\theta_b} &= -T_{m_{left}} + T_{f_{left}} - T_{m_{right}}+T_{f_{right}} + l_b \, F \, cos(\theta_b) \\
      &= -Q_{\theta_{w_{left}}}-Q_{\theta_{w_{right}}} + l_b \, F \, cos(\theta_b)
    \end{aligned}
  }

\section{Linearization}

\[
  \mathrm{
    \begin{aligned}
      & ( m_b \, l_b^2 + I_{b_{yy}} ) \, \ddot{\theta}_b + \frac{m_b \, l_b \, l_w}{2} \, ( \ddot{\theta}_{w_{left}} + \ddot{\theta}_{w_{right}} ) - m_b \, l_b \, g \, \theta_b + \\
      & + 2 \, ( \frac{K_e \, K_t}{R_m} + b_f ) \, \dot{\theta}_b - ( \frac{K_e \, K_t}{R_m} + b_f ) \, ( \dot{\theta}_{w_{left}} + \dot{\theta}_{w_{right}} ) = \\
      & = - \frac{K_t}{R_m} \, ( V_{left} + V_{right} ) + ( l_b - l_w ) \, F
    \end{aligned}
  }
\]

\[
  \mathrm{
    \begin{aligned}
      & \frac{m_b \, l_w^2}{4} \, ( \ddot{\theta}_{w_{left}} + \ddot{\theta}_{w_{right}} ) + \frac{m_b \, l_b \, l_w}{2} \, \ddot{\theta}_b - \frac{l_w^2}{L^2} \, ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) \, I_{b_{zz}} + \\
      & + m_w \, l_w^2 \, \ddot{\theta}_{w_{left}} + I_{w_{xx}} \, \ddot{\theta}_{w_{left}} - \frac{2 \, I_{w_{zz}} \, l_w^2}{L^2} \, ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) + \\
      & + \frac{K_e \, K_t}{R_m} \, ( \dot{\theta}_{w_{left}} - \dot{\theta}_b ) + b_f \, ( \dot{\theta}_{w_{left}} - \dot{\theta}_b ) = \frac{K_t}{R_m} \, V_{left} + \frac{l_w}{2} \, F
    \end{aligned}
  }
\]

\[
  \mathrm{
    \begin{aligned}
      & \frac{m_b \, l_w^2}{4} \, ( \ddot{\theta}_{w_{left}} + \ddot{\theta}_{w_{right}} ) + \frac{m_b \, l_b \, l_w}{2} \, \ddot{\theta}_b + \frac{l_w^2}{L^2} \, ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) \, I_{b_{zz}} + \\
      & + m_w \, l_w^2 \, \ddot{\theta}_{w_{right}} + I_{w_{xx}} \, \ddot{\theta}_{w_{right}} + \frac{2 \, I_{w_{zz}} \, l_w^2}{L^2} \, ( \ddot{\theta}_{w_{right}} - \ddot{\theta}_{w_{left}} ) + \\
      & + \frac{K_e \, K_t}{R_m} \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_b ) + b_f \, ( \dot{\theta}_{w_{right}} - \dot{\theta}_b ) = \frac{K_t}{R_m} \, V_{right} + \frac{l_w}{2} \, F
    \end{aligned}
  }
\]
